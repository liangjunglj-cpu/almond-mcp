"""Model the modular kit the v016 districts are assembled from, in Blender.

One kit, three atmospheres. The districts differ by layout, palette, lighting and weather, not by
geometry, so the kit has to carry enough variety to read as a market, a flooded service level and a
container yard. Every module is built at metre scale with its origin at the base centre and its pivot on
a grid multiple, so the district builder can place them without per-piece fudging.

Modules fall into four families:

* ground     -- 8 m slabs: plain, cracked, drain, and a kerbed edge
* structure  -- 4 m walls: blank, shuttered, shopfront, pipe-covered; plus a pillar and a stair flight
* market     -- stall counter, awning, crate stack, produce bin, hanging sign bar
* industrial -- shipping container and its open variant, pipe run, catwalk, floodlight mast, fence panel,
                bollard, barrel, air handler, antenna mast

Run:
    blender --background --factory-startup --python scripts/build_district_kit_v016.py -- --out <dir>
"""
from __future__ import annotations

CONCRETE = "TFKitConcrete"
METAL = "TFKitMetal"
PAINT = "TFKitPaint"
GLASS = "TFKitGlass"
EMISSIVE = "TFKitEmissive"


def reset() -> None:
    bpy.ops.wm.read_factory_settings(use_empty=True)
    for name, base, rough, metal in (
        (CONCRETE, (0.42, 0.43, 0.45, 1.0), 0.78, 0.0),
        (METAL, (0.34, 0.35, 0.38, 1.0), 0.42, 1.0),
        (PAINT, (0.62, 0.24, 0.26, 1.0), 0.5, 0.0),
        (GLASS, (0.12, 0.16, 0.2, 1.0), 0.12, 0.0),
        (EMISSIVE, (0.2, 0.9, 0.85, 1.0), 0.3, 0.0),
    ):
        material = bpy.data.materials.new(name)
        material.use_nodes = True
        principled = next(n for n in material.node_tree.nodes if n.type == "BSDF_PRINCIPLED")
        principled.inputs["Base Color"].default_value = base
        principled.inputs["Roughness"].default_value = rough
        principled.inputs["Metallic"].default_value = metal
        if name == EMISSIVE:
            principled.inputs["Emission Color"].default_value = base
            principled.inputs["Emission Strength"].default_value = 4.0


def box(name, size, location=(0, 0, 0), rotation=(0, 0, 0), bevel=0.0, segments=2, material=CONCRETE):
    bpy.ops.mesh.primitive_cube_add(size=1.0, location=location, rotation=rotation)
    obj = bpy.context.active_object
    obj.name = name
    obj.scale = size
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
    if bevel > 0.0:
        modifier = obj.modifiers.new("Bevel", "BEVEL")
        modifier.width = bevel
        modifier.segments = segments
        modifier.limit_method = "ANGLE"
        bpy.context.view_layer.objects.active = obj
        bpy.ops.object.modifier_apply(modifier=modifier.name)
    assign(obj, material)
    return obj


def tube(name, radius, depth, location=(0, 0, 0), rotation=(0, 0, 0), verts=16, material=METAL):
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=location,
                                        rotation=rotation, vertices=verts)
    obj = bpy.context.active_object
    obj.name = name
    assign(obj, material)
    return obj


def assign(obj, material_name) -> None:
    obj.data.materials.clear()
    obj.data.materials.append(bpy.data.materials[material_name])


def finish(name, parts):
    bpy.ops.object.select_all(action="DESELECT")
    for part in parts:
        part.select_set(True)
    bpy.context.view_layer.objects.active = parts[0]
    if len(parts) > 1:
        bpy.ops.object.join()
    obj = bpy.context.view_layer.objects.active
    obj.name = name
    # The join keeps the active part's own transform, so bake it in before measuring or exporting;
    # without this a module whose first part was rotated ships rotated.
    bpy.ops.object.transform_apply(location=False, rotation=True, scale=True)
    bpy.ops.object.origin_set(type="ORIGIN_GEOMETRY", center="BOUNDS")
    obj.location = (0, 0, 0)
    bpy.context.view_layer.update()
    lowest = min(Vector(c).z for c in obj.bound_box)
    for vertex in obj.data.vertices:
        vertex.co.z -= lowest
    obj.data.update()
    metre_uv(obj)
    return obj


def metre_uv(obj) -> None:
    """Planar-project each face on its dominant axis so one UV unit is one metre, as the v009 kit does."""
    layer = obj.data.uv_layers.get("UVMap") or obj.data.uv_layers.new(name="UVMap")
    for polygon in obj.data.polygons:
        axis = max(range(3), key=lambda k: abs(polygon.normal[k]))
        others = [i for i in range(3) if i != axis]
        for loop_index in polygon.loop_indices:
            co = obj.data.vertices[obj.data.loops[loop_index].vertex_index].co
            layer.data[loop_index].uv = (co[others[0]], co[others[1]])


def scatter_panel_lines(obj, count, seed, depth=0.02):
    """Cut shallow horizontal grooves so a flat wall catches a highlight instead of reading as a slab."""
    import random
    rng = random.Random(seed)
    mesh = bmesh.new()
    mesh.from_mesh(obj.data)
    faces = [f for f in mesh.faces if abs(f.normal.z) < 0.4]
    for _ in range(count):
        if not faces:
            break
        face = rng.choice(faces)
        try:
            result = bmesh.ops.inset_individual(mesh, faces=[face], thickness=rng.uniform(0.04, 0.12),
                                                depth=-depth)
            faces = [f for f in mesh.faces if abs(f.normal.z) < 0.4]
            del result
        except Exception:
            break
    mesh.to_mesh(obj.data)
    mesh.free()

def ground_plain():

    parts = [box("Slab", (8.0, 8.0, 0.24), bevel=0.02)]

def container(open_door=False):
    parts = []
    body = box("Body", (6.06, 2.44, 2.59), bevel=0.03, material=PAINT)
    parts.append(body)
    for index in range(18):
        rib = box(f"Rib{index}", (0.1, 2.46, 2.3), location=(-2.9 + index * 0.34, 0, 0), material=PAINT)
        parts.append(rib)
    for x in (-3.03, 3.03):
        end = box(f"End{x}", (0.06, 2.46, 2.61), location=(x, 0, 0), material=METAL)
        parts.append(end)
    for x in (-1, 1):
        for y in (-1, 1):
            for z in (-1, 1):

                             location=(x * 2.88, y * 1.07, z * 1.15), bevel=0.03, material=METAL)
                parts.append(corner)
    if open_door:
        for side in (-1, 1):
            door = box(f"Door{side}", (0.08, 1.15, 2.4), location=(3.2, side * 1.5, 0),
                       rotation=(0, 0, math.radians(side * -62)), material=PAINT)
            parts.append(door)

        for side in (-1, 1):

                    rotation=(0, 0, angle), material=METAL)
        parts.append(blade)
    grille = box("Grille", (0.06, 1.2, 0.7), location=(-0.98, 0, 0.05), material=METAL)
    parts.append(grille)
    for index in range(7):
        louvre = box(f"Louvre{index}", (0.1, 1.2, 0.05), location=(-1.0, 0, -0.26 + index * 0.1),
                     rotation=(0, math.radians(22), 0), material=METAL)
        parts.append(louvre)
    for x in (-1, 1):
        for y in (-1, 1):
            foot = box(f"Foot{x}{y}", (0.2, 0.2, 0.14), location=(x * 0.8, y * 0.6, -0.55),
                       material=METAL)
            parts.append(foot)
    return finish("SM_KitAirHandler", parts), "1.9 m air handler"


def antenna_mast():
    parts = [box("Base", (0.8, 0.8, 0.2), bevel=0.03, material=METAL)]
    mast = tube("Mast", 0.08, 5.0, location=(0, 0, 2.6), verts=12)
    parts.append(mast)
    for index, z in enumerate((1.6, 2.8, 4.0)):
        span = 0.9 - index * 0.22
        cross = box(f"Cross{index}", (span, 0.05, 0.05), location=(0, 0, z), material=METAL)
        parts.append(cross)
        for side in (-1, 1):
            element = tube(f"Element{index}{side}", 0.025, 0.5,
                           location=(side * span * 0.45, 0, z + 0.25), verts=8)
            parts.append(element)
    dish = tube("Dish", 0.45, 0.06, location=(0.3, 0, 4.7), rotation=(0, math.radians(58), 0),
                verts=20, material=METAL)
    parts.append(dish)
    beacon = tube("Beacon", 0.07, 0.12, location=(0, 0, 5.16), verts=10, material=EMISSIVE)
    parts.append(beacon)
    return finish("SM_KitAntennaMast", parts), "5 m antenna mast with a beacon"


MODULES = [
    ground_plain, ground_cracked, ground_drain, ground_kerb,
    wall_blank, wall_shutter, wall_shopfront, wall_pipes, pillar, stair_flight,
    stall_counter, stall_awning, crate_stack, produce_bin, sign_bar,
    lambda: container(False), lambda: container(True),
    pipe_run, catwalk, floodlight_mast, fence_panel, bollard, barrel, air_handler, antenna_mast,
]


def triangles(obj) -> int:
    return sum(len(p.vertices) - 2 for p in obj.data.polygons)


def export(obj, path: Path) -> None:
    bpy.ops.object.select_all(action="DESELECT")
    obj.select_set(True)
    bpy.context.view_layer.objects.active = obj
    bpy.ops.export_scene.fbx(
        filepath=str(path), use_selection=True, object_types={"MESH"},
        axis_forward="-Y", axis_up="Z", apply_unit_scale=True,
        apply_scale_options="FBX_SCALE_UNITS", bake_anim=False,
        mesh_smooth_type="FACE", path_mode="STRIP",
    )


def main() -> None:
    argv = sys.argv[sys.argv.index("--") + 1:] if "--" in sys.argv else []
    parser = argparse.ArgumentParser()
    parser.add_argument("--out", required=True)
    args = parser.parse_args(argv)
    out = Path(args.out)
    out.mkdir(parents=True, exist_ok=True)
    records = []
    for builder in MODULES:
        reset()
        obj, note = builder()
        path = out / f"{obj.name}.fbx"
        export(obj, path)
        bpy.context.view_layer.update()
        records.append({
            "asset": obj.name, "file": path.name, "note": note,
            "triangles": triangles(obj),
            "dimensions_m": [round(v, 3) for v in obj.dimensions],
            "materials": [m.name for m in obj.data.materials],
        })
        print(f"{obj.name}: {triangles(obj)} tris, {[round(v,2) for v in obj.dimensions]} m")
    (out / "kit.json").write_text(json.dumps({"kit": "district_v016", "modules": records}, indent=1),
                                  encoding="utf-8")
    print(f"\n{len(records)} modules -> {out}")


if __name__ == "__main__":
    main()
